        global_status = str(global_result.get("status", "")).upper()

    return status in {"FAILED", "ERROR"}

    try:
        spark.table(table_name).limit(0).count()
        return True
    except Exception:
        return False


def actual_col(df, wanted, required=False):
    by_lower = {c.lower(): c for c in df.columns}
    found = by_lower.get(wanted.lower())
    if required and not found:
        raise ValueError(f"Column {wanted} was not found. Available columns: {df.columns}")
    return found


def col_or_null(df, wanted):
    found = actual_col(df, wanted, required=False)
    return F.col(found) if found else F.lit(None)


def load_bronze(bronze_table):
    if not table_exists(bronze_table):
        raise ValueError(f"Bronze table does not exist: {bronze_table}")
    df = spark.table(bronze_table)
    ingest_col = actual_col(df, "IngestRunId", required=False)
    if p_ingest_run_id and ingest_col:
        df = df.where(F.col(ingest_col) == F.lit(p_ingest_run_id))
    return df


def get_task_config(method_name, config_id=None):

    if config_id is None:
        config_id = p_config_id

    task = (
        spark.table("meta.TaskConfig")
             .filter(
                 (F.col("ConfigId") == config_id) &
                 (F.col("Method") == method_name) &
                 (F.col("IsActive") == 1)
             )
             .select(
                 "SourceTable",
                 "TargetSchema",
                 "TargetTable",
                 "ExecutionOrder",
                 "LoadType",
                 "TaskConfigId"
             )
             .first()
    )

    if task is None:
        raise Exception(
            f"No active TaskConfig found for Method={method_name}, ConfigId={config_id}"
        )

    return task.asDict()

def target_columns_for(src_df, silver_table, configured_target_columns=None, transforms=None, drop_columns=None):
    transforms = transforms or {}
    drop = {c.lower() for c in (drop_columns or [])}
    if table_exists(silver_table):
        return spark.table(silver_table).columns
    if configured_target_columns:
        return configured_target_columns
    cols = [c for c in src_df.columns if c.lower() not in METADATA_COLUMNS and c.lower() not in drop]
    for c in transforms:
        if c not in cols:
            cols.append(c)
    return cols


def align_to_target(src_df, silver_table, configured_target_columns=None, transforms=None, drop_columns=None):
    transforms = transforms or {}
    cols = target_columns_for(src_df, silver_table, configured_target_columns, transforms, drop_columns)
    target_schema = {}
    if table_exists(silver_table):
        target_schema = {f.name: f.dataType for f in spark.table(silver_table).schema.fields}

    exprs = []
    for target_col in cols:
        if target_col in transforms:
            expr = transforms[target_col]
        else:
            source_col = actual_col(src_df, target_col, required=False)
            expr = F.col(source_col) if source_col else F.lit(None).cast("string")
        if target_col in target_schema:
            expr = expr.cast(target_schema[target_col])
        exprs.append(expr.alias(target_col))
    return src_df.select(*exprs)


def null_safe_condition(keys):
    return " AND ".join([f"target.`{k}` <=> source.`{k}`" for k in keys])


def checksum_changed_condition(checksum_col):
    return f"NOT (target.`{checksum_col}` <=> source.`{checksum_col}`)"


def checksum_same_condition(checksum_col):
    return f"target.`{checksum_col}` <=> source.`{checksum_col}`"


def pre_reset_sites(silver_table, source_df, flag_col, active_value, inactive_value):
    if not table_exists(silver_table) or "SiteCode" not in source_df.columns:
        return 0
    sites = [r[0] for r in source_df.select("SiteCode").where(F.col("SiteCode").isNotNull()).distinct().collect()]
    if not sites:
        return 0
    before = spark.table(silver_table).where((F.col("SiteCode").isin(sites)) & (F.col(flag_col) == F.lit(active_value))).count()
    if before:
        DeltaTable.forName(spark, silver_table).update(
            condition=(F.col("SiteCode").isin(sites)) & (F.col(flag_col) == F.lit(active_value)),
            set={flag_col: F.lit(inactive_value)},
        )
    return before


def apply_legacy_services_insert_scope(source_df, silver_table, match_keys):
    # Legacy SaveServices does not insert a brand-new sID when that site already has service rows.
    if not table_exists(silver_table) or "SiteCode" not in source_df.columns:
        return source_df, 0

    sites = [r[0] for r in source_df.select("SiteCode").where(F.col("SiteCode").isNotNull()).distinct().collect()]
    if not sites:
        return source_df, 0

    target_df = spark.table(silver_table)
    existing_sites = target_df.where(F.col("SiteCode").isin(sites)).select("SiteCode").distinct().withColumn("_site_has_services", F.lit(1))
    target_keys = target_df.select(*match_keys).dropDuplicates().withColumn("_matched_service_key", F.lit(1))

    scoped = source_df.join(existing_sites, "SiteCode", "left").join(target_keys, match_keys, "left")
    skipped = scoped.where(F.col("_site_has_services").isNotNull() & F.col("_matched_service_key").isNull()).count()
    allowed = scoped.where(F.col("_site_has_services").isNull() | F.col("_matched_service_key").isNotNull())
    return allowed.drop("_site_has_services", "_matched_service_key"), skipped


# CDC -- Start
def process_cdc(df, match_keys):
    """
    SQL Server CDC Operations

    1 = Delete
    2 = Insert
    3 = Before Update (Ignore)
    4 = After Update
    """

    df = df.filter(F.col("__$operation") != 3)

    window = (
        Window.partitionBy(*match_keys)
        .orderBy(
            F.col("__$start_lsn").desc(),
            F.col("__$seqval").desc(),
            F.col("__$command_id").desc(),
            F.col("__$operation").desc()
        )
    )

    return (
        df.withColumn("_rn", F.row_number().over(window))
          .filter(F.col("_rn") == 1)
          .drop("_rn")
    )


def merge_to_silver_cdc(
    method_name,
    match_keys,
    configured_target_columns=None,
    transforms=None,
    drop_columns=None,
    pre_reset=None,
    preserve_legacy_services_insert_scope=False,
):
    if bronze_failed_for(method_name):
        return result_payload(
            method_name,
            "SKIPPED",
            message="Bronze method failed, so silver was skipped."
        )

    rows_read = 0
    rows_inserted = 0
    rows_updated = 0
    rows_deleted = 0
    rows_skipped = 0

    task = get_task_config(method_name)

    print("=" * 80)
    print(f"Method        : {method_name}")
    print(f"TaskConfigId  : {task['TaskConfigId']}")
    print(f"ExecutionOrder: {task['ExecutionOrder']}")
    print(f"Bronze Table  : {task['SourceTable']}")
    print(f"Silver Table  : {task['TargetSchema']}.{task['TargetTable']}")
    print("=" * 80)

    bronze_table = task["SourceTable"]
    target_schema = task["TargetSchema"]
    target_table = task["TargetTable"]

    silver_table = f"{target_schema}.{target_table}"

    try:

        bronze_df = load_bronze(bronze_table)

        rows_read = bronze_df.count()

        # ==========================================================
        # Process CDC first
        # ==========================================================

        cdc_df = process_cdc(
            bronze_df,
            match_keys
        )

        if rows_read == 0:

            if not table_exists(silver_table):

                empty_df = align_to_target(
                    bronze_df,
                    silver_table,
                    configured_target_columns,
                    transforms,
                    drop_columns,
                )

                for key in match_keys:
                    actual_col(empty_df, key, required=True)

                empty_df.limit(0) \
                    .write \
                    .format("delta") \
                    .mode("overwrite") \
                    .saveAsTable(silver_table)

            return result_payload(
                method_name,
                "SUCCESS",
                rows_read=0,
                message="No bronze rows found for this ingest run."
            )

        # ==========================================================
        # Split CDC while CDC metadata still exists
        # ==========================================================
        
        delete_df = (
            cdc_df
            .filter(F.col("__$operation") == 1)
        )
        
        update_df = (
            cdc_df
            .filter(F.col("__$operation") == 4)
        )
        
        insert_df = (
            cdc_df
            .filter(F.col("__$operation") == 2)
        )

        rows_updated = update_df.count()

        rows_deleted = delete_df.count()

        rows_inserted = insert_df.count()


        print("=" * 80)
        print("CDC AFTER process_cdc()")
        print("=" * 80)
        
        cdc_df.groupBy("__$operation").count().show()
        
        print("=" * 80)
        print("DELETE ROWS")
        print("=" * 80)
        
        delete_df.select(
            "SiteCode",
            "LiqId",
            "__$start_lsn",
            "__$seqval",
            "__$command_id"
        ).show(100, False)

        print("=" * 80)
        print("UPDATE ROWS")
        print("=" * 80)
        
        update_df.select(
            "SiteCode",
            "LiqId",
            "__$start_lsn",
            "__$seqval",
            "__$command_id"
        ).show(100, False)
        
        print("=" * 80)
        print("INSERT ROWS")
        print("=" * 80)
        
        insert_df.select(
            "SiteCode",
            "LiqId",
            "__$start_lsn"
        ).show(100, False)
        
        # ==========================================================
        # Remove CDC columns only now
        # ==========================================================
        
        delete_df = align_to_target(
            delete_df,
            silver_table,
            configured_target_columns,
            transforms,
            drop_columns,
        ).cache()

        update_df = align_to_target(
            update_df,
            silver_table,
            configured_target_columns,
            transforms,
            drop_columns,
        ).cache()
        
        insert_df = align_to_target(
            insert_df,
            silver_table,
            configured_target_columns,
            transforms,
            drop_columns,
        ).cache()
        
        target_exists = table_exists(silver_table)

        if pre_reset and target_exists:
            rows_skipped += pre_reset_sites(
                silver_table,
                update_df,
                pre_reset["flag_col"],
                pre_reset["active_value"],
                pre_reset["inactive_value"],
            )
            
        if preserve_legacy_services_insert_scope:
            insert_df, legacy_skipped = apply_legacy_services_insert_scope(
                insert_df,
                silver_table,
                match_keys,
            )
            rows_skipped += legacy_skipped


        rows_after_scope = (
            rows_updated +
            rows_deleted +
            rows_inserted
        )

        if rows_after_scope == 0:
        
            if not target_exists:
            
                delete_df.unionByName(update_df).unionByName(insert_df).limit(0) \
                    .write \
                    .format("delta") \
                    .mode("overwrite") \
                    .saveAsTable(silver_table)
            delete_df.unpersist()
            update_df.unpersist()
                    
            insert_df.unpersist()
            return result_payload(
                method_name,
                "SUCCESS",
                rows_read=rows_read,
                rows_skipped=rows_skipped,
                message="No rows remained after legacy scope rules."
            )

        if not target_exists:
        
            delete_df.unionByName(update_df).unionByName(insert_df) \
                .write \
                .format("delta") \
                .mode("overwrite") \
                .saveAsTable(silver_table)

            delete_df.unpersist()
            update_df.unpersist()
            insert_df.unpersist()

            return result_payload(
                method_name,
                "SUCCESS",
                rows_read=rows_read,
                rows_inserted=rows_after_scope,
                rows_skipped=rows_skipped,
            )
        target_columns = update_df.columns

        update_all = {
            c: f"source.`{c}`"
            for c in target_columns
        }

        insert_all = {
            c: f"source.`{c}`"
            for c in target_columns
        }

        # ==========================================================
        # UPDATE / DELETE MERGE
        # ==========================================================

        if rows_updated + rows_deleted > 0:

            min_liqid_df = (
                delete_df.unionByName(update_df)
                .groupBy("SiteCode")
                .agg(
                    F.min("LiqId").alias("MinLiqId")
                )
            )

            pruning_conditions = []

            for row in min_liqid_df.collect():

                pruning_conditions.append(
                    f"(target.SiteCode = '{row['SiteCode']}' "
                    f"AND target.LiqId >= {row['MinLiqId']})"
                )

            merge_condition = (
                "("
                + " OR ".join(pruning_conditions)
                + ") AND ("
                + null_safe_condition(match_keys)
                + ")"
            )

            print("=" * 80)
            print("Executing Pruned Delete Merge")
            print("=" * 80)
            print(merge_condition)

            (
                DeltaTable.forName(spark, silver_table)
                .alias("target")
                .merge(
                    delete_df.alias("source"),
                    merge_condition
                )
                .whenMatchedDelete()
                .execute()
            )

            print("=" * 80)
            print("Executing Pruned Update Merge")
            print("=" * 80)
            print(merge_condition)

            (
                DeltaTable.forName(spark, silver_table)
                .alias("target")
                .merge(
                    update_df.alias("source"),
                    merge_condition
                )
                .whenMatchedUpdate(
                    set=update_all
                )
                .execute()
            )


        else:

            print("=" * 80)
            print("No Update/Delete rows found.")
            print("Skipping Pruned Merge.")
            print("=" * 80)

        # ==========================================================
        # INSERT MERGE
        # ==========================================================

        if rows_inserted > 0:

            print("=" * 80)
            print("Executing Insert Merge")
            print("=" * 80)

            (
                DeltaTable.forName(spark, silver_table)
                .alias("target")
                .merge(
                    insert_df.alias("source"),
                    null_safe_condition(match_keys)
                )
                .whenNotMatchedInsert(
                    values=insert_all
                )
                .execute()
            )

        else:

            print("=" * 80)
            print("No Insert rows found.")
            print("Skipping Insert Merge.")
            print("=" * 80)

        # ==========================================================
        # UPDATE LSN
        # ==========================================================

        if rows_after_scope > 0:

            print(f"Rows After Scope: {rows_after_scope}")

            site_lsns = (
                cdc_df
                .groupBy("SiteCode")
                .agg(
                    F.max("__$start_lsn").alias("LatestLSN")
                )
                .collect()
            )

            for row in site_lsns:

                print(f"Updating Site: {row['SiteCode']}")
                print(f"Latest LSN   : {row['LatestLSN']}")

                spark.sql(f"""
                    UPDATE meta.TaskConfig
                    SET
                        ScheduleExpression = '{row["LatestLSN"]}',
                        LastRunDate = current_timestamp()
                    WHERE
                        ConfigId = {p_bronze_config_id}
                        AND Method = '{method_name}'
                        AND SiteCode = '{row["SiteCode"]}'
                        AND IsActive = 1
                """)

        delete_df.unpersist()
        update_df.unpersist()
        insert_df.unpersist()

        return result_payload(
            method_name,
            "SUCCESS",
            rows_read=rows_read,
            rows_inserted=rows_inserted,
            rows_updated=rows_updated,
            rows_deleted=rows_deleted,
            rows_skipped=rows_skipped,
        )

    except Exception as ex:

        if 'delete_df' in locals():
            delete_df.unpersist()

        if 'update_df' in locals():
            update_df.unpersist()

        if 'insert_df' in locals():
            insert_df.unpersist()

        if is_notebook_exit(ex):
            raise

        return result_payload(
            method_name,
            "FAILED",
            rows_read=rows_read,
            rows_inserted=rows_inserted,
            rows_updated=rows_updated,
            rows_deleted=rows_deleted,
            rows_skipped=rows_skipped,
            message=traceback.format_exc(),
        )

# CDC -- End

def merge_to_silver(
    method_name,
    match_keys,
    configured_target_columns=None,
    transforms=None,
    drop_columns=None,
    checksum_col=None,
    same_checksum_update_columns=None,
    pre_reset=None,
    insert_only_columns=None,
    preserve_legacy_services_insert_scope=False,
):
    if bronze_failed_for(method_name):
        return result_payload(
            method_name,
            "SKIPPED",
            message="Bronze method failed, so silver was skipped."
        )    

    rows_read = 0
    rows_inserted = 0
    rows_updated = 0
    rows_skipped = 0

    task = get_task_config(method_name)

    print("=" * 80)
    print(f"Method        : {method_name}")
    print(f"TaskConfigId  : {task['TaskConfigId']}")
    print(f"ExecutionOrder: {task['ExecutionOrder']}")
    print(f"Bronze Table  : {task['SourceTable']}")
    print(f"Silver Table  : {task['TargetSchema']}.{task['TargetTable']}")
    print("=" * 80)
    
    bronze_table = task["SourceTable"]
    target_schema = task["TargetSchema"]
    target_table = task["TargetTable"]

    silver_table = f"{target_schema}.{target_table}"

    try:
        bronze_df = load_bronze(bronze_table)
        rows_read = bronze_df.count()
        if rows_read == 0:
            if not table_exists(silver_table):
                empty_df = align_to_target(bronze_df, silver_table, configured_target_columns, transforms, drop_columns)
                for key in match_keys:
                    actual_col(empty_df, key, required=True)
                empty_df.limit(0).write.format("delta").mode("overwrite").saveAsTable(silver_table)
            return result_payload(method_name, "SUCCESS", rows_read=0, message="No bronze rows found for this ingest run.")

        silver_df = align_to_target(bronze_df, silver_table, configured_target_columns, transforms, drop_columns)
        for key in match_keys:
            actual_col(silver_df, key, required=True)
        silver_df = silver_df.dropDuplicates(match_keys)

        target_exists = table_exists(silver_table)
        if pre_reset and target_exists:
            rows_skipped += pre_reset_sites(
                silver_table,
                silver_df,
                pre_reset["flag_col"],
                pre_reset["active_value"],
                pre_reset["inactive_value"],
            )

        if preserve_legacy_services_insert_scope:
            silver_df, legacy_skipped = apply_legacy_services_insert_scope(silver_df, silver_table, match_keys)
            rows_skipped += legacy_skipped

        rows_after_scope = silver_df.count()
        if rows_after_scope == 0:
            if not target_exists:
                silver_df.limit(0).write.format("delta").mode("overwrite").saveAsTable(silver_table)
            return result_payload(method_name, "SUCCESS", rows_read=rows_read, rows_skipped=rows_skipped, message="No rows remained after legacy scope rules.")

        if not target_exists:
            silver_df.write.format("delta").mode("overwrite").saveAsTable(silver_table)
            rows_inserted = rows_after_scope
            return result_payload(method_name, "SUCCESS", rows_read=rows_read, rows_inserted=rows_inserted, rows_skipped=rows_skipped)

        target_keys = spark.table(silver_table).select(*match_keys).dropDuplicates()
        rows_inserted = silver_df.join(target_keys, match_keys, "left_anti").count()
        rows_updated = silver_df.join(target_keys, match_keys, "inner").count()

        insert_only = set(insert_only_columns or [])
        update_columns = [c for c in silver_df.columns if c not in insert_only]
        update_all = {c: f"source.`{c}`" for c in update_columns}
        insert_all = {c: f"source.`{c}`" for c in silver_df.columns}

        merge_builder = DeltaTable.forName(spark, silver_table).alias("target").merge(
            silver_df.alias("source"),
            null_safe_condition(match_keys),
        )

        if checksum_col and same_checksum_update_columns:
            same_update = {c: f"source.`{c}`" for c in same_checksum_update_columns}
            merge_builder = merge_builder.whenMatchedUpdate(
                condition=checksum_changed_condition(checksum_col),
                set=update_all,
            ).whenMatchedUpdate(
                condition=checksum_same_condition(checksum_col),
                set=same_update,
            )
        elif checksum_col:
            merge_builder = merge_builder.whenMatchedUpdate(
                condition=checksum_changed_condition(checksum_col),
                set=update_all,
            )
        else:
            merge_builder = merge_builder.whenMatchedUpdate(set=update_all)

        merge_builder.whenNotMatchedInsert(values=insert_all).execute()
        return result_payload(method_name, "SUCCESS", rows_read=rows_read, rows_inserted=rows_inserted, rows_updated=rows_updated, rows_skipped=rows_skipped)

    except Exception as ex:
        if is_notebook_exit(ex):
            raise
        return result_payload(method_name, "FAILED", rows_read=rows_read, rows_inserted=rows_inserted, rows_updated=rows_updated, rows_skipped=rows_skipped, message=traceback.format_exc())
