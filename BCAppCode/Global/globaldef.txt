def with_delta_retry(func, max_retries=3):
    attempt = 0
    while attempt < max_retries:
        try:
            return func()
        except Exception as ex:
            error = str(ex)
            if any(x in error for x in ["Concurrent", "MetadataChanged", "Delta"]):
                attempt += 1
                time.sleep(5)
            else:
                raise
    raise Exception(f"Failed after {max_retries} retries")

def with_delta_retry(func, max_retries=3):
    attempt = 0
    while attempt < max_retries:
        try:
            return func()
        except Exception as ex:
            error = str(ex)
            if any(x in error for x in ["Concurrent", "MetadataChanged", "Delta"]):
                attempt += 1
                time.sleep(5)
            else:
                raise
    raise Exception(f"Failed after {max_retries} retries")

def bronze_succeeded(method_name):
    """Skip table when ADF/pipeline marks the Bronze method as failed."""
    return _method_key(method_name) not in bronze_failed_methods

def get_active_locations():

    return loc_df.select(col("SId"), col("SiteCode").alias("_LocSiteCode"))

def read_bronze_batch(bronze_table):
    bronze_df = spark.table(bronze_table)
    
    # Filter by run_date if it exists in Bronze
    if "run_date" in bronze_df.columns:
        bronze_df = bronze_df.filter(to_date(col("run_date")) == lit(business_date))
        
    # FORCE overwrite run_id so Silver and Gold match perfectly
    bronze_df = bronze_df.withColumn("run_id", lit(run_id))
        
    rows_read = bronze_df.count()
    if rows_read == 0:
        print(f"No source rows found in {bronze_table} for business_date {business_date}.")
    
    return bronze_df, rows_read

def write_silver_batch(silver_df, silver_table):
    rows_written = silver_df.count()
    if rows_written == 0:
        return 0

    silver_df = silver_df.persist()

    with_delta_retry(
        lambda: silver_df.write
            .format("delta")
            .mode("overwrite")
            .option("mergeSchema", "false")
            .saveAsTable(silver_table)
    )

    silver_df.unpersist()

    return rows_written

# Legacy BAM date parsing: strip prefix before ", " then parse
def parse_bam_date(date_col_name):
    cleaned = expr(f"trim(case when instr({date_col_name}, ', ') > 0 then substring({date_col_name}, instr({date_col_name}, ', ') + 2, length({date_col_name})) else {date_col_name} end)")
    return coalesce(to_timestamp(cleaned), to_timestamp(col(date_col_name)), col(date_col_name))

def get_location_map():
    return spark.table(LOCATION_CONS_TABLE).select(col("SiteCode").alias("loc_SiteCode"), "ConnectionId")

def resolve_sites_in_df(df, site_col="SiteCode"):
    loc_df = get_location_map()
    return df.join(broadcast(loc_df), df[site_col] == loc_df["loc_SiteCode"], "left").withColumn(
        f"Normalized{site_col}",
        when(df[site_col] == "B59A", "AdvMD-NF")
        .when(col("ConnectionId") == 2, "SAMMS")
        .when(col("ConnectionId") == 4, "AdvMD-AR")
        .otherwise(df[site_col])
    ).drop("ConnectionId", "loc_SiteCode")

def get_change_condition(batch_df, key_cols, exclude_cols):
    if "RowChkSum" in batch_df.columns:
        return col("t.RowChkSum") != col("s.RowChkSum")
        
    compare_cols = [c for c in batch_df.columns if c not in key_cols and c not in exclude_cols]
    if not compare_cols:
        return lit(True)
        
    change_condition = ~col(f"t.{compare_cols[0]}").eqNullSafe(col(f"s.{compare_cols[0]}"))
    for c in compare_cols[1:]:
        change_condition = change_condition | ~col(f"t.{c}").eqNullSafe(col(f"s.{c}"))
    return change_condition

def process_payer(silver_table, active_gold_table, version_gold_table):
    silver_df, rows_read = read_silver_batch(silver_table)
    
    silver_df = resolve_sites_in_df(silver_df, "SiteCode")
    
    batch = silver_df.drop("SiteCode").withColumnRenamed("NormalizedSiteCode", "SiteCode").drop("run_date")
    batch = batch.withColumn("RowState", lit(1).cast("int")).withColumn("LastModAt", current_timestamp())
    batch = batch.dropDuplicates(["SiteCode", "PayId"])
    batch = batch.persist(StorageLevel.MEMORY_AND_DISK)
    
    current_sitecodes = None
    existing_gold = None
    existing_gold_current_sites = None
    other_site_records = None
    current_keys = None
    
    try:
        current_sitecodes = batch.select("SiteCode").distinct().persist(StorageLevel.MEMORY_AND_DISK)
        existing_gold_raw = get_existing_gold(active_gold_table, batch.schema)
        
        gold_window = Window.partitionBy("SiteCode", "PayId").orderBy(col("LastModAt").desc_nulls_last())
        existing_gold = (
            existing_gold_raw.withColumn("rn", row_number().over(gold_window))
            .filter(col("rn") == 1).drop("rn")
            .persist(StorageLevel.MEMORY_AND_DISK)
        )
        
        existing_gold_current_sites = existing_gold.alias("g").join(
            broadcast(current_sitecodes.alias("s")), col("g.SiteCode") == col("s.SiteCode"), "inner"
        ).select("g.*").persist(StorageLevel.MEMORY_AND_DISK)
        
        other_site_records = existing_gold.alias("g").join(
            broadcast(current_sitecodes.alias("s")), col("g.SiteCode") == col("s.SiteCode"), "left_anti"
        ).persist(StorageLevel.MEMORY_AND_DISK)
        
        current_keys = batch.select("SiteCode", "PayId").distinct().persist(StorageLevel.MEMORY_AND_DISK)
        
        unchanged_df = (
            existing_gold_current_sites.alias("t")
            .join(batch.alias("s"), on=["SiteCode", "PayId"], how="inner")
            .filter((col("t.RowChkSum") == col("s.RowChkSum")) & (col("t.RowState") != 0))
            .select("t.*")
        )
        
        updated_df = (
            batch.alias("s")
            .join(existing_gold_current_sites.alias("t"), on=["SiteCode", "PayId"], how="inner")
            .filter((col("t.RowChkSum") != col("s.RowChkSum")) | (col("t.RowState") == 0))
            .select("s.*")
        )
        
        new_records_df = (
            batch.alias("s")
            .join(existing_gold_current_sites.alias("t"), on=["SiteCode", "PayId"], how="left_anti")
            .select("s.*")
        )
        
        inactive_df = (
            existing_gold_current_sites.alias("g")
            .join(current_keys.alias("s"), on=["SiteCode", "PayId"], how="left_anti")
            .filter(col("g.RowState") != lit(0).cast("int"))
            .select("g.*")
            .withColumn("RowState", lit(0).cast("int"))
            .withColumn("LastModAt", current_timestamp())
            .withColumn("run_id", lit(run_id))
        )
        
        already_inactive_df = (
            existing_gold_current_sites.alias("g")
            .join(current_keys.alias("s"), on=["SiteCode", "PayId"], how="left_anti")
            .filter(col("g.RowState") == lit(0).cast("int"))
            .select("g.*")
        )
        
        final_gold_df = other_site_records.unionByName(unchanged_df, allowMissingColumns=True) \
            .unionByName(updated_df, allowMissingColumns=True) \
            .unionByName(new_records_df, allowMissingColumns=True) \
            .unionByName(inactive_df, allowMissingColumns=True) \
            .unionByName(already_inactive_df, allowMissingColumns=True)
            
        final_gold_df = final_gold_df.withColumn("LastModAt", coalesce(col("LastModAt"), current_timestamp()))
        rows_written = final_gold_df.count()
        
        if rows_written > 0:
            final_gold_df.write.option(Constants.WorkspaceId, gold_workspace_id).option(Constants.DatawarehouseId, gold_warehouse_id).mode("overwrite").synapsesql(version_gold_table)
                
    finally:
        batch.unpersist()
        if current_sitecodes: current_sitecodes.unpersist()
        if existing_gold: existing_gold.unpersist()
        if existing_gold_current_sites: existing_gold_current_sites.unpersist()
        if other_site_records: other_site_records.unpersist()
        if current_keys: current_keys.unpersist()
        
    return rows_read, rows_written

    current_keys = None
    
    try:
        existing_gold_raw = get_existing_gold(active_gold_table, batch.schema)
        
        gold_window = Window.partitionBy("Id").orderBy(col("LastModAt").desc_nulls_last())

            existing_gold_raw.withColumn("rn", row_number().over(gold_window))
            .filter(col("rn") == 1).drop("rn")
            .persist(StorageLevel.MEMORY_AND_DISK)
        )
        
        current_keys = batch.select("Id").distinct().persist(StorageLevel.MEMORY_AND_DISK)
        
        change_cond = get_change_condition(batch, ["Id"], ["LastModAt", "run_id"])
        
        unchanged_df = (
            existing_gold.alias("t")
            .join(batch.alias("s"), on=["Id"], how="inner")
            .filter(~change_cond)
            .select("t.*")
        )
        
        updated_df = (
            batch.alias("s")
            .join(existing_gold.alias("t"), on=["Id"], how="inner")
            .filter(change_cond)
            .select("s.*")
        )
        
        new_records_df = (
            batch.alias("s")
            .join(existing_gold.alias("t"), on=["Id"], how="left_anti")
            .select("s.*")
        )
        
        inactive_df = (
            existing_gold.alias("g")
            .join(current_keys.alias("s"), on=["Id"], how="left_anti")
            .filter(col("g.RowState") == True)
            .select("g.*")
            .withColumn("RowState", lit(False))
            .withColumn("LastModAt", current_timestamp())
            .withColumn("run_id", lit(run_id))
        )
        
        already_inactive_df = (
            existing_gold.alias("g")
            .join(current_keys.alias("s"), on=["Id"], how="left_anti")
            .filter(col("g.RowState") == False)
            .select("g.*")
        )
        
        final_gold_df = unchanged_df.unionByName(updated_df, allowMissingColumns=True) \
            .unionByName(new_records_df, allowMissingColumns=True) \
            .unionByName(inactive_df, allowMissingColumns=True) \
            .unionByName(already_inactive_df, allowMissingColumns=True)
            
        final_gold_df = final_gold_df.withColumn("LastModAt", coalesce(col("LastModAt"), current_timestamp()))
        rows_written = final_gold_df.count()
        
        if rows_written > 0:
            final_gold_df.write.option(Constants.WorkspaceId, gold_workspace_id).option(Constants.DatawarehouseId, gold_warehouse_id).mode("overwrite").synapsesql(version_gold_table)
                
    finally:
        batch.unpersist()
        if existing_gold: existing_gold.unpersist()
        if current_keys: current_keys.unpersist()
        
    return rows_read, rows_written

def get_change_condition(batch_df, key_cols, exclude_cols):
    if "RowChkSum" in batch_df.columns:
        return col("t.RowChkSum") != col("s.RowChkSum")
        
    compare_cols = [c for c in batch_df.columns if c not in key_cols and c not in exclude_cols]
    if not compare_cols:
        return lit(True)
        
    change_condition = ~col(f"t.{compare_cols[0]}").eqNullSafe(col(f"s.{compare_cols[0]}"))
    for c in compare_cols[1:]:
        change_condition = change_condition | ~col(f"t.{c}").eqNullSafe(col(f"s.{c}"))
    return change_condition
